Example implementation outline:
- from django.conf import settings
- def generate_username(email, claims=None):
    if claims is None:
        claims = {}
    claim_name = getattr(settings, "OIDC_USERNAME_CLAIM", "email")
    candidate = claims.get(claim_name) or email
    return unicodedata.normalize("NFKC", str(candidate))[:150]
