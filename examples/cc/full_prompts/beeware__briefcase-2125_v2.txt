   def install_apk(self, apk_path: str | Path):
       try:
           self.run("install", "-r", apk_path)
       except subprocess.CalledProcessError as e:
           parts = []
           for attr in ("output", "stderr"):
               val = getattr(e, attr, None)
               if val:
                   if isinstance(val, bytes):
                       try:
                           val = val.decode(errors="replace")
                       except Exception:
                           val = str(val)
                   parts.append(str(val))
           failing_output = "\n".join(p for p in parts if p).strip()
           if failing_output:
               console = getattr(self.tools, "console", None)
               try:
                   if console and hasattr(console, "error"):
                       # You may include a brief preface; tests will check for the adb message itself.
                       for line in failing_output.splitlines():
                           console.error(line)
                   else:
                       print(failing_output)
               except Exception:
                   pass
           raise BriefcaseCommandError(
               f"Unable to install APK {apk_path} on {self.device}"
           ) from e

   import pytest
   from briefcase.exceptions import BriefcaseCommandError

   def test_install_failure_shows_subprocess_output(adb, capsys, monkeypatch):
       # Simulate adb failing with a meaningful error message.
       # ADB.run uses tools.subprocess.check_output, which populates CalledProcessError.output on failure.
       error = subprocess.CalledProcessError(
           returncode=1,
           cmd="adb install -r example.apk",
           output="Failure [INSTALL_FAILED_UPDATE_INCOMPATIBLE: Package com.example.helloworld signatures do not match previously installed version]"
       )
       # Make adb.run raise the error when called.
       def raise_error(*args, **kwargs):
           raise error
       monkeypatch.setattr(adb, "run", raise_error)

       with pytest.raises(BriefcaseCommandError):
           adb.install_apk("example.apk")

       captured = capsys.readouterr()
       combined = (captured.out or "") + (captured.err or "")
       assert "INSTALL_FAILED_UPDATE_INCOMPATIBLE" in combined
