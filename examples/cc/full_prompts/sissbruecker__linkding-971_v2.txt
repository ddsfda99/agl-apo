import unicodedata
from django.conf import settings

def _get_claim_value(claims, name):
    if not claims:
        return None
    # Try mapping-like get(name)
    try:
        return claims.get(name)
    except Exception:
        pass
    # Try attribute access
    try:
        return getattr(claims, name)
    except Exception:
        return None

def generate_username(email, claims=None):
    """
    Generate a Django username from OIDC claims with a configurable claim name,
    falling back to the email parameter. Normalizes to NFKC and truncates to 150 chars.
    Compatible with mozilla-django-oidc which calls this with (email, claims).
    """
    claim_name = getattr(settings, "OIDC_USERNAME_CLAIM", "email")
    candidate = _get_claim_value(claims, claim_name)
    if not candidate:
        candidate = email
    candidate = "" if candidate is None else str(candidate)
    return unicodedata.normalize("NFKC", candidate)[:150]

Notes:
- The signature generate_username(email, claims=None) matches mozilla-django-oidc expectations and tolerates being called with a single argument.
- Claims can be dicts or objects; the helper attempts claims.get(name) first and falls back to getattr(claims, name) to be resilient across library versions.
