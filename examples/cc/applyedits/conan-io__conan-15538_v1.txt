import io
from contextlib import redirect_stdout

# Import the text formatter under test
from conan.cli.commands.graph import cli_build_order

def _run_and_capture(func, *args, **kwargs):
    buf = io.StringIO()
    with redirect_stdout(buf):
        func(*args, **kwargs)
    return [line for line in buf.getvalue().splitlines() if line.strip()]

def test_cli_build_order_recipe_mode_prints_levels():
    # Recipe-mode shape: items include 'packages' as levels (list of lists)
    info = [
        {
            'ref': 'pkg/1.0',
            'packages': [
                [  # level 0
                    {'package_id': 'pid1', 'binary': 'Build'},
                ],
                [  # level 1
                    {'package_id': 'pid2', 'binary': 'Cache'},
                    {'package_id': 'pid3', 'binary': 'Build'},
                ],
            ]
        }
    ]

    lines = _run_and_capture(cli_build_order, info)

    # It should print one line per package entry across levels
    assert 'pkg/1.0:pid1 - Build' in lines
    assert 'pkg/1.0:pid2 - Cache' in lines
    assert 'pkg/1.0:pid3 - Build' in lines

def test_cli_build_order_configuration_mode_prints_one_line_per_item():
    # Configuration-mode shape: flat items without 'packages'
    info = [
        {'ref': 'ffmpeg/6.1', 'package_id': 'abc123', 'binary': 'Build'},
        {'ref': 'zlib/1.2.13', 'package_id': 'def456', 'binary': 'Cache'},
    ]

    lines = _run_and_capture(cli_build_order, info)

    expected = {'ffmpeg/6.1:abc123 - Build', 'zlib/1.2.13:def456 - Cache'}

    # Exactly one line per input item, matching the expected text
    assert set(lines) == expected
