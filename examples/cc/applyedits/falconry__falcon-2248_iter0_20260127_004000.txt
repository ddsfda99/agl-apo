# At module import time: backport Partitioned attribute and auto-add logic
def _ensure_partitioned_cookie_support():
    try:
        from http import cookies as _cookies
    except Exception:
        return

    M = getattr(_cookies, 'Morsel', None)
    if M is None:
        return

    # Add 'partitioned' to reserved map and flags, if missing
    reserved = getattr(M, '_reserved', None)
    if isinstance(reserved, dict) and 'partitioned' not in reserved:
        reserved['partitioned'] = 'Partitioned'

    flags = getattr(M, '_flags', None)
    if isinstance(flags, set) and 'partitioned' not in flags:
        flags.add('partitioned')

    # Monkey-patch __setitem__ to auto-add Partitioned when appropriate
    if not getattr(M, '_falcon_partitioned_patched', False):
        _orig_setitem = M.__setitem__

        def _patched_setitem(self, key, value):
            # First perform the original set
            _orig_setitem(self, key, value)

            # Then auto-add Partitioned if SameSite=None and Secure is set
            try:
                samesite = self.get('samesite')
                is_none = isinstance(samesite, str) and samesite.lower() == 'none'
                is_secure = bool(self.get('secure'))
                has_partitioned = bool(self.get('partitioned'))
                if is_none and is_secure and not has_partitioned:
                    _orig_setitem(self, 'partitioned', True)
            except Exception:
                # Be conservative: never raise from here
                pass

        M.__setitem__ = _patched_setitem
        M._falcon_partitioned_patched = True

# Run the patch at import time
_ensure_partitioned_cookie_support()
del _ensure_partitioned_cookie_support
