Goal

import subprocess
import pytest

from briefcase.exceptions import BriefcaseCommandError

def test_install_failure_shows_subprocess_output(adb, capsys, monkeypatch):
    # Simulate adb failing with a meaningful error message.
    # The adb.run wrapper would normally call tools.subprocess.check_output,
    # which populates CalledProcessError.output on failure.
    error = subprocess.CalledProcessError(
        returncode=1,
        cmd="adb install -r example.apk",
        output="Failure [INSTALL_FAILED_UPDATE_INCOMPATIBLE: Package com.example.helloworld signatures do not match previously installed version]"
    )
    monkeypatch.setattr(adb, "run", lambda *a, **kw: (_ for _ in ()).throw(error))

    with pytest.raises(BriefcaseCommandError):
        adb.install_apk("example.apk")

    captured = capsys.readouterr()
    # Validate that the failing adb output is printed to console.
    # Check both stdout and stderr to be robust.
    combined = (captured.out or "") + (captured.err or "")
    assert "INSTALL_FAILED_UPDATE_INCOMPATIBLE" in combined

In src/briefcase/integrations/android_sdk.py, within class ADB:

    def install_apk(self, apk_path: str | Path):
        try:
            self.run("install", "-r", apk_path)
        except subprocess.CalledProcessError as e:
            # Extract any output from the failing subprocess and show it to the user.
            # tools.subprocess.check_output usually places combined stdout/stderr into e.output.
            # Be defensive if output is bytes or if stderr is used.
            parts = []
            for attr in ("output", "stderr"):
                val = getattr(e, attr, None)
                if val:
                    if isinstance(val, bytes):
                        try:
                            val = val.decode(errors="replace")
                        except Exception:
                            val = str(val)
                    parts.append(str(val))

            failing_output = "\n".join(p for p in parts if p).strip()

            if failing_output:
                # Prefer console if available; otherwise fall back to plain print.
                try:
                    console = getattr(self.tools, "console", None)
                    if console and hasattr(console, "error"):
                        console.error("adb output from failed install:")
                        for line in failing_output.splitlines():
                            console.error(f"    {line}")
                    else:
                        print("adb output from failed install:")
                        print(failing_output)
                except Exception:
                    # In case console printing fails for any reason, still proceed to raise.
                    pass

            raise BriefcaseCommandError(
                f"Unable to install APK {apk_path} on {self.device}"
            ) from e

- Only prints in the error path.
- Shows the exact adb output so users can search for the message.
