from django.conf import settings
import unicodedata

def generate_username(email, claims=None):
    if claims is None:
        claims = {}
    claim_name = getattr(settings, "OIDC_USERNAME_CLAIM", "email")
    candidate = None
    if isinstance(claims, dict) and claim_name in claims:
        candidate = claims.get(claim_name)
    if not candidate:
        candidate = email
    candidate = "" if candidate is None else str(candidate)
    return unicodedata.normalize("NFKC", candidate)[:150]

Notes:
