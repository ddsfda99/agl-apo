  import importlib.util
  import sys
  import types
  import os
  from pathlib import Path
  import importlib.metadata as metadata

  def load_base_module(fake_default_dir):
      # Stub out platformdirs so tests do not require the real dependency.
      fake_pd = types.ModuleType("platformdirs")
      class PlatformDirs:
          def __init__(self, module_name, roaming):
              self.user_data_dir = fake_default_dir
      fake_pd.PlatformDirs = PlatformDirs
      sys.modules["platformdirs"] = fake_pd

      # Patch importlib.metadata.version to avoid PackageNotFoundError.
      orig_version = metadata.version
      metadata.version = lambda name: "0.0"

      # Patch os.walk so Templates class sees a non-empty templates directory,
      # without touching the repository filesystem.
      orig_os_walk = os.walk
      def fake_walk(path):
          yield (str(path), ["fake_template"], [])
      os.walk = fake_walk

      try:
          base_path = Path("reflex/constants/base.py").resolve()
          spec = importlib.util.spec_from_file_location("reflex_constants_base", str(base_path))
          module = importlib.util.module_from_spec(spec)
          spec.loader.exec_module(module)
          return module
      finally:
          # Restore patched symbols and cleanup stubbed module.
          os.walk = orig_os_walk
          metadata.version = orig_version
          sys.modules.pop("platformdirs", None)

  def test_env_overrides_dir():
      os.environ["REFLEX_DIR"] = "/tmp/reflexdir"
      try:
          base = load_base_module("/tmp/default")
          assert base.Reflex.DIR == "/tmp/reflexdir"
      finally:
          os.environ.pop("REFLEX_DIR", None)

  def test_default_uses_platformdirs():
      os.environ.pop("REFLEX_DIR", None)
      base = load_base_module("/tmp/defaultdir")
      assert base.Reflex.DIR == "/tmp/defaultdir"
